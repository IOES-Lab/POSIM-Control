cmake_minimum_required(VERSION 3.22)
project(posim_control LANGUAGES CXX)
include(CTest)
if(BUILD_TESTING)
  add_executable(following-test tests/FollowingTest.cc)
  target_compile_features(following-test PRIVATE cxx_std_17)
  target_compile_options(following-test PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/UNDEBUG> $<$<NOT:$<CXX_COMPILER_ID:MSVC>>:-UNDEBUG>)
  add_test(NAME following COMMAND following-test)
endif()
option(POSIM_CONTROL_BUILD_PLUGINS "Build Gazebo system plugins" ON)
if(POSIM_CONTROL_BUILD_PLUGINS)
find_package(gz-sim REQUIRED)
find_package(gz-plugin REQUIRED COMPONENTS register loader)
add_library(wwos-promotion-follower SHARED Follower.cc)
target_compile_features(wwos-promotion-follower PRIVATE cxx_std_17)
target_link_libraries(wwos-promotion-follower PRIVATE gz-sim::gz-sim gz-plugin::register)
install(TARGETS wwos-promotion-follower LIBRARY DESTINATION lib)
add_library(wwos-promotion-pilot SHARED Pilot.cc)
target_compile_features(wwos-promotion-pilot PRIVATE cxx_std_17)
target_link_libraries(wwos-promotion-pilot PRIVATE gz-sim::gz-sim gz-plugin::register)

set_target_properties(wwos-promotion-pilot wwos-promotion-follower PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
install(TARGETS wwos-promotion-pilot LIBRARY DESTINATION lib)
endif()
